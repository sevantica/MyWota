mifare.security.hmac_key=5A5B5C5D5E5F606162636465666768696A6B6C6D6E6F707172737475767778

# ----------------------------------------------------------------------------
# Key Derivation Settings
# ----------------------------------------------------------------------------
# Number of PBKDF2 iterations for key derivation
# Higher = more secure but slower card init (~300ms per 1000 iterations)
# EMBEDDED RECOMMENDATION: 1000-2000 (acceptable security, fast init)
# Server systems: 10000+ (better security, longer init acceptable)
# Min: 1000, Max: 100000, Recommended: 1000 (embedded), 10000 (server)
mifare.security.pbkdf2_iterations=1000

# ----------------------------------------------------------------------------
# Custom Sector Keys (Optional - Advanced)
# ----------------------------------------------------------------------------
# Use custom MIFARE sector keys instead of factory defaults (0xFF...)
# When enabled, prevents generic readers from accessing your cards
# 0=Disabled (use factory keys), 1=Enabled (use custom keys below)
mifare.security.use_custom_sector_keys=0

# mifare.security.sector4_key_a=FFFFFFFFFFFF
# mifare.security.sector4_key_b=FFFFFFFFFFFF

# ----------------------------------------------------------------------------
# Security Feature Enables
# ----------------------------------------------------------------------------
# Enable HMAC-SHA256 authentication tags (recommended)
# Detects any tampering with card data
mifare.security.enable_hmac_auth=1

# Enable replay protection with rolling counters (recommended)
# Prevents rollback attacks (using old card backups)
mifare.security.enable_replay_protection=1

# Enable challenge-response protocol (advanced - optional)
# Prevents card cloning - requires extra card blocks
# Set to 1 only if you need maximum anti-cloning protection
mifare.security.enable_challenge_response=0

# Maximum clock drift allowed in seconds (24 hours = 86400)
# Used for timestamp validation in replay protection
mifare.security.max_timestamp_drift_sec=86400

# Failed challenge attempts before card lockout (when challenge enabled)
mifare.security.failed_challenge_lockout=5

# ----------------------------------------------------------------------------
# Block-Level Encryption Selection
# ----------------------------------------------------------------------------
# Choose which blocks to encrypt (all recommended for maximum security)

# Encrypt user data blocks 5, 6 (token count, status, state)
mifare.security.encrypt_user_data=1

# Encrypt transaction log blocks 9, 10 (transaction history)
mifare.security.encrypt_transactions=1

# Encrypt token cache blocks 13, 14 (fast balance access)
mifare.security.encrypt_token_cache=1

# Encrypt account data block 16 (phone number, validity)
mifare.security.encrypt_account_data=1

# ============================================================================
# Security Architecture Overview
# ============================================================================
# 
# Layer 1: Custom MIFARE Keys
#   - Optional sector keys prevent unauthorized readers
#   - Two-tier access: Key A (read), Key B (write)
#
# Layer 2: AES-128-CTR Encryption
#   - Per-card unique keys derived from UID + master secret
#   - Per-block unique IVs prevent pattern analysis
#   - PBKDF2 key derivation with 10,000 iterations
#
# Layer 3: HMAC-SHA256 Authentication
#   - 4-byte truncated HMAC per encrypted block
#   - Detects any bit-flipping or tampering
#   - Cryptographically secure (can't forge without key)
#
# Layer 4: Challenge-Response (Optional)
#   - Prevents card cloning with challenge protocol
#   - Random nonce + HMAC response validation
#   - Card lockout after failed attempts
#
# Layer 5: Replay Protection
#   - Rolling write counters detect rollback
#   - Timestamp validation with drift tolerance
#   - Session nonces prevent replay attacks
#
# ============================================================================
# Performance Impact
# ============================================================================
# - Card Init: +50ms (key derivation)
# - Card Read: +2-5ms per encrypted block
# - Card Write: +2-5ms per encrypted block
# - Total Overhead: ~20-30ms per transaction (acceptable for dispenser)
